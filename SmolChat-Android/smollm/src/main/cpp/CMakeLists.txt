cmake_minimum_required(VERSION 3.22.1)
project("smollm")

# Additive, opt-in Vulkan-enabled ggml backend (see build_library_vulkan()
# below and vendor/README.md). Off by default, so the existing CPU-only
# build below is completely unaffected: Gradle only ever passes this ON
# from a separate product flavor, which builds in its own directory and
# never touches the default flavor's build.
option(SMOLLM_BUILD_VULKAN_VARIANT "Build an additional Vulkan-enabled smollm variant (arm64-v8a only)" OFF)
if (SMOLLM_BUILD_VULKAN_VARIANT)
    set(SMOLLM_VENDOR_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
    # glslc (the GLSL -> SPIR-V shader compiler ggml-vulkan needs at build
    # time) ships inside the NDK but isn't on PATH or auto-discovered by
    # find_package(Vulkan). CMAKE_ANDROID_NDK/ANDROID_HOST_TAG are set by
    # the NDK's own toolchain file, already processed by this point.
    set(Vulkan_GLSLC_EXECUTABLE "${CMAKE_ANDROID_NDK}/shader-tools/${ANDROID_HOST_TAG}/glslc" CACHE FILEPATH "" FORCE)
    # SPIRV-Headers never registers a build-tree package (no export(PACKAGE
    # ...)), so find_package(... CONFIG) needs a real installed-looking
    # prefix; vendor/spirv-headers is exactly that layout, pre-generated
    # once rather than adding a nested install step to this build.
    set(SPIRV-Headers_DIR "${SMOLLM_VENDOR_DIR}/spirv-headers/share/cmake/SPIRV-Headers" CACHE PATH "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    # ggml-vulkan is built as its own shared library (separate from
    # ggml-base/ggml), which only self-registers via ggml's dynamic-backend
    # discovery mechanism (ggml_backend_load() / ggml_backend_load_all(),
    # see LLMInference.cpp) -- and that mechanism only works if the loaded
    # .so actually exports the `ggml_backend_init` entrypoint symbol it
    # looks for. That symbol is generated by the GGML_BACKEND_DL_IMPL macro
    # (see ggml-backend-impl.h), which is a no-op unless this option is on.
    # Confirmed via a real on-device probe: without it, ggml_backend_load()
    # dlopen()s libggml-vulkan.so successfully but then logs "failed to
    # find ggml_backend_init" and gives up -- not a guess.
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
endif()

# Additive, opt-in OpenCL-enabled ggml backend for Adreno GPUs (see
# vendor/README.md). Off by default, same isolation guarantee as
# SMOLLM_BUILD_VULKAN_VARIANT above: a separate product flavor, a separate
# build directory, never touches the default CPU-only build.
option(SMOLLM_BUILD_OPENCL_VARIANT "Build an additional OpenCL-enabled smollm variant (arm64-v8a only)" OFF)
if (SMOLLM_BUILD_OPENCL_VARIANT)
    if (NOT DEFINED SMOLLM_VENDOR_DIR)
        set(SMOLLM_VENDOR_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
    endif()
    # find_package(OpenCL REQUIRED) inside ggml-opencl/CMakeLists.txt uses
    # CMake's own FindOpenCL module, which does a real find_path()/
    # find_library() search -- but skips that search and trusts these
    # values as-is when they're already present as CACHE entries (which
    # FORCE-setting them here guarantees), exactly like Vulkan_GLSLC_EXECUTABLE
    # and SPIRV-Headers_DIR above short-circuit their own lookups.
    set(OpenCL_INCLUDE_DIR "${SMOLLM_VENDOR_DIR}/opencl-headers" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY "${SMOLLM_VENDOR_DIR}/opencl-icd-loader/libOpenCL.so" CACHE FILEPATH "" FORCE)
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    # Same lesson as Vulkan above, applied from the start this time rather
    # than discovered the hard way: ggml-opencl is its own shared library,
    # so it only self-registers through ggml's dynamic-backend discovery
    # (ggml_backend_load(), see LLMInference.cpp), which needs the
    # ggml_backend_init entrypoint symbol that only GGML_BACKEND_DL_IMPL
    # generates -- a no-op unless this is on.
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
endif()

add_subdirectory(../../../../llama.cpp llama.cpp)
set(LLAMA_DIR_RELATIVE "../../../../llama.cpp")
get_filename_component(LLAMA_DIR ${LLAMA_DIR_RELATIVE} ABSOLUTE)

set(GGML_DIR ${LLAMA_DIR}/ggml)
set(COMMON_DIR ${LLAMA_DIR}/common)
set(VENDOR_DIR ${LLAMA_DIR}/vendor)

# vulkan/vulkan.hpp (the C++ bindings ggml-vulkan.cpp includes) and
# spirv/unified1/spirv.hpp aren't NDK-provided; only the ggml-vulkan target
# itself needs them, so this doesn't touch any CPU-only target's flags.
if (SMOLLM_BUILD_VULKAN_VARIANT AND TARGET ggml-vulkan)
    target_include_directories(ggml-vulkan PRIVATE
            ${SMOLLM_VENDOR_DIR}/vulkan-hpp
            ${SMOLLM_VENDOR_DIR}/spirv-headers/include
    )
endif()

# The vendored libOpenCL.so (see vendor/README.md) is a real prebuilt
# binary, not something this configure pass builds -- so unlike every other
# .so here, Gradle's externalNativeBuild packaging step (which just scans
# CMAKE_LIBRARY_OUTPUT_DIRECTORY for whatever landed there after the build)
# would never see it on its own. Copying it there ourselves at configure
# time is what gets it included in the APK alongside libggml-opencl.so.
#
# (Tried removing this + relying on uses-native-library in
# AndroidManifest.xml to grant access to the device's real vendor
# libOpenCL.so instead -- confirmed the app then hangs indefinitely rather
# than working or failing cleanly, see conversation. Reverted; bundling our
# own shim here remains the working, if OpenCL-non-functional, behavior.)
if (SMOLLM_BUILD_OPENCL_VARIANT AND TARGET ggml-opencl)
    file(COPY ${SMOLLM_VENDOR_DIR}/opencl-icd-loader/libOpenCL.so
         DESTINATION ${CMAKE_LIBRARY_OUTPUT_DIRECTORY})
endif()

# llama-common is built by upstream llama.cpp as a SHARED library whenever
# BUILD_SHARED_LIBS is on (the Android default), but its headers (common.h,
# chat.h, ...) declare functions with no export macro (unlike llama.h's
# LLAMA_API), so with -fvisibility=hidden those symbols never get exported
# from libllama-common.so, and linking against it fails at link time.
# Rather than patching upstream headers, compile llama-common's own sources
# into our own STATIC library here, so its symbols link directly into each
# libsmollm*.so and never cross a shared-library visibility boundary.
# llama and ggml are left untouched and continue to build SHARED as before.
file(GLOB_RECURSE LLAMA_COMMON_STATIC_SOURCES CONFIGURE_DEPENDS ${COMMON_DIR}/*.cpp)
add_library(llama-common-static STATIC ${LLAMA_COMMON_STATIC_SOURCES})
target_include_directories(
        llama-common-static
        PUBLIC
        ${COMMON_DIR}
        ${VENDOR_DIR}
        ${LLAMA_DIR}/include
        ${GGML_DIR}/include
        ${GGML_DIR}/src
        ${GGML_DIR}/src/ggml-cpu
)
target_compile_features(llama-common-static PUBLIC cxx_std_17)
target_compile_options(llama-common-static PUBLIC -fvisibility=hidden -fvisibility-inlines-hidden)
set_target_properties(llama-common-static PROPERTIES POSITION_INDEPENDENT_CODE ON)
if (LLAMA_SUBPROCESS)
    target_compile_definitions(llama-common-static PUBLIC LLAMA_SUBPROCESS)
endif()
target_link_libraries(llama-common-static PUBLIC llama-common-base)
target_link_libraries(llama-common-static PRIVATE cpp-httplib)

# -fvisibility=hidden: hide all symbols by default
# -fvisibility-inlines-hidden: hide all inline symbols by default
target_compile_options(
        llama
        PUBLIC
        -fvisibility=hidden -fvisibility-inlines-hidden
)
# -ffunction-sections: place each function in its own section
# -fdata-sections: place each data member in its own section
target_compile_options(
        llama
        PUBLIC
        -ffunction-sections -fdata-sections
)
target_link_options(
        llama
        PRIVATE
        -Wl,--gc-sections -flto
        -Wl,--exclude-libs,ALL
)

# compiling for different CPU extensions for Arm64 (aarch64)
# See docs/build_arm_flags.md for more details

function(build_library target_name)
    add_library(
            ${target_name}
            SHARED
            LLMInference.cpp
            smollm.cpp
    )
    target_include_directories(
            ${target_name}
            PUBLIC
            ${COMMON_DIR}
            ${GGML_DIR}/include
            ${GGML_DIR}/src
            ${GGML_DIR}/src/ggml-cpu
            ${LLAMA_DIR}/include
            ${VENDOR_DIR}
    )
    # -fvisibility=hidden: hide all symbols by default
    # -fvisibility-inlines-hidden: hide all inline symbols by default
    target_compile_options(
            ${target_name}
            PUBLIC
            -fvisibility=hidden -fvisibility-inlines-hidden
    )
    # -ffunction-sections: place each function in its own section
    # -fdata-sections: place each data member in its own section
    target_compile_options(
            ${target_name}
            PUBLIC
            -ffunction-sections -fdata-sections
    )
    target_link_libraries(
            ${target_name}
            android log llama llama-common-static vulkan
    )
    # -Wl,--gc-sections: remove unused sections (garbage collection)
    # -flto: link-time optimization
    # -Wl,--exclude-libs,ALL: exclude all libraries
    target_link_options(
            ${target_name}
            PRIVATE
            -Wl,--gc-sections -flto
            -Wl,--exclude-libs,ALL
    )
endfunction()


function(build_library_arm64 target_name cpu_flags)
    build_library(${target_name})
    set(GGML_SYSTEM_ARCH "ARM")
    set(GGML_CPU_KLEIDIAI ON)
    set(GGML_OPENMP ON)
    target_compile_definitions(${target_name} PRIVATE
            GGML_SYSTEM_ARCH=${GGML_SYSTEM_ARCH}
            GGML_CPU_KLEIDIAI=$<BOOL:${GGML_CPU_KLEIDIAI}>
            GGML_OPENMP=$<BOOL:${GGML_OPENMP}>
    )
    target_compile_options(
            ${target_name}
            PUBLIC
            -DGGML_USE_CPU -DGGML_USE_CPU_AARCH64 ${cpu_flags} -O3
    )
endfunction()

function(build_library_vulkan target_name cpu_flags)
    # Vulkan support itself comes entirely from GGML_VULKAN being set globally
    # for this configure pass (see SMOLLM_BUILD_VULKAN_VARIANT above), which
    # links the ggml-vulkan backend straight into libggml.so; this function's
    # own body only sets the same CPU baseline the other arm64 targets use,
    # named separately so its purpose is explicit in the target list below.
    build_library_arm64(${target_name} ${cpu_flags})
endfunction()

function(build_library_opencl target_name cpu_flags)
    # Same shape as build_library_vulkan above: OpenCL support comes from
    # GGML_OPENCL being set globally for this configure pass (see
    # SMOLLM_BUILD_OPENCL_VARIANT above); this function's own body just sets
    # the same CPU baseline the other arm64 targets use.
    build_library_arm64(${target_name} ${cpu_flags})
endfunction()

function(build_library_armv7a target_name cpu_flags fpu fpu_abi)
    build_library(${target_name})
    target_compile_options(
            ${target_name}
            PUBLIC
            -DGGML_USE_CPU ${cpu_flags} ${fpu} ${fpu_abi} -O3
    )
endfunction()

function(build_library_universal target_name)
    build_library(${target_name})
    target_compile_options(
            ${target_name}
            PUBLIC
            -DGGML_USE_CPU -O3
    )
endfunction()

if (SMOLLM_BUILD_VULKAN_VARIANT)
    # Separate flavor, separate build directory, separate APK (see
    # vendor/README.md) - deliberately not also building the CPU-only
    # target lineup below in this same configure pass, since GGML_VULKAN is
    # a single global option for the whole ggml/llama target graph in this
    # invocation; keeping this pass Vulkan-only is what keeps the default
    # flavor's own (separate) CMake invocation, and its ggml/llama targets,
    # completely untouched.
    if (${ANDROID_ABI} STREQUAL "arm64-v8a")
        build_library_vulkan("smollm_vulkan" "-march=armv8-a")
    endif()
elseif (SMOLLM_BUILD_OPENCL_VARIANT)
    # Same isolation reasoning as the Vulkan branch above, for GGML_OPENCL.
    if (${ANDROID_ABI} STREQUAL "arm64-v8a")
        build_library_opencl("smollm_opencl" "-march=armv8-a")
    endif()
else()
    build_library_universal("smollm")
    if (${ANDROID_ABI} STREQUAL "armeabi-v7a")
        build_library_armv7a("smollm_v7a" "-march=armv7-a" "-mfpu=neon-vfpv4" "-mfloat-abi=softfp")
    endif()
    if (${ANDROID_ABI} STREQUAL "arm64-v8a")
        build_library_arm64("smollm_v8" "-march=armv8-a")
        # Targets for Arm-v8.2a
        build_library_arm64("smollm_v8_2_fp16" "-march=armv8.2-a+fp16")
        build_library_arm64("smollm_v8_2_fp16_dotprod" "-march=armv8.2-a+fp16+dotprod")

        # Targets for Arm-v8.4a
        build_library_arm64("smollm_v8_4_fp16_dotprod" "-march=armv8.4-a+fp16+dotprod")
        build_library_arm64("smollm_v8_4_fp16_dotprod_sve" "-march=armv8.4-a+fp16+dotprod+sve")
        build_library_arm64("smollm_v8_4_fp16_dotprod_i8mm" "-march=armv8.4-a+fp16+dotprod+i8mm")
        build_library_arm64("smollm_v8_4_fp16_dotprod_i8mm_sve" "-march=armv8.4-a+fp16+dotprod+i8mm+sve")
    endif()
endif()
